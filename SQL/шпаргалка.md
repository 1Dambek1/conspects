1️⃣ Простая выборка
from sqlalchemy import select
from sqlalchemy.orm import Session

with Session(engine) as session:
    # Все пользователи
    stmt = select(User)
    users = session.execute(stmt).scalars().all()

2️⃣ Фильтрация (WHERE)
# Пользователи с именем, содержащим "Ан"
stmt = select(User).where(User.name.like("%Ан%"))
users = session.execute(stmt).scalars().all()

3️⃣ Сортировка (ORDER BY)
# Продукты по цене по убыванию
stmt = select(Product).order_by(Product.price.desc())
products = session.execute(stmt).scalars().all()

4️⃣ Ограничение количества (LIMIT)
# 5 самых дорогих продуктов
stmt = select(Product).order_by(Product.price.desc()).limit(5)
products = session.execute(stmt).scalars().all()

5️⃣ JOIN и фильтрация по связям
# Все заказы пользователя с user_id = 1
stmt = select(Order).where(Order.user_id == 1)
orders = session.execute(stmt).scalars().all()

# JOIN для фильтрации: заказы с товарами дороже 100
stmt = (
    select(Order)
    .join(Order.items)
    .join(OrderItem.product)
    .where(Product.price > 100)
)
orders = session.execute(stmt).scalars().all()

6️⃣ Агрегации
from sqlalchemy import func

# Количество пользователей
count_users = session.execute(select(func.count(User.id))).scalar_one()

# Общая сумма заказа (quantity * price)
stmt = (
    select(Order.id, func.sum(OrderItem.quantity * Product.price).label("total_sum"))
    .join(OrderItem)
    .join(Product)
    .group_by(Order.id)
)
result = session.execute(stmt).all()

7️⃣ Среднее, максимум, минимум
# Средняя стоимость заказа
subquery = (
    select(Order.id, func.sum(OrderItem.quantity * Product.price).label("order_total"))
    .join(OrderItem)
    .join(Product)
    .group_by(Order.id)
    .subquery()
)
avg_order = session.execute(select(func.round(func.avg(subquery.c.order_total), 2))).scalar_one()

8️⃣ Пользователи без заказов
stmt = select(User).outerjoin(Order).where(Order.id == None)
users_without_orders = session.execute(stmt).scalars().all()

9️⃣ Продукты, которые ни разу не были заказаны
stmt = select(Product).outerjoin(OrderItem).where(OrderItem.id == None)
products_not_ordered = session.execute(stmt).scalars().all()

🔟 Количество проданных единиц по продуктам
stmt = (
    select(
        Product.id,
        Product.name,
        func.coalesce(func.sum(OrderItem.quantity), 0).label("total_sold")
    )
    .outerjoin(OrderItem)
    .group_by(Product.id, Product.name)
)
result = session.execute(stmt).all()

1️⃣1️⃣ Использование selectinload для загрузки связанных объектов
from sqlalchemy.orm import selectinload

# Заказы с товарами и их информацией
stmt = (
    select(Order)
    .options(
        selectinload(Order.items).selectinload(OrderItem.product)
    )
)
orders = session.execute(stmt).scalars().all()

for order in orders:
    for item in order.items:
        print(item.product.name, item.quantity)

🔹 Полезные функции
| Функция                         | Назначение                         |
| ------------------------------- | ---------------------------------- |
| `func.count()`                  | подсчет записей                    |
| `func.sum()`                    | суммирование                       |
| `func.avg()`                    | среднее значение                   |
| `func.round(value, n)`          | округление до n знаков             |
| `func.coalesce(value, default)` | замена NULL на default             |
| `.scalar()` / `.scalar_one()`   | возвращает одно значение           |
| `.scalars()`                    | возвращает генератор ORM-объектов  |
| `.join()`                       | объединение таблиц (INNER JOIN)    |
| `.outerjoin()`                  | LEFT JOIN                          |
| `.group_by()`                   | группировка по колонкам            |
| `.where()`                      | фильтрация данных                  |
| `.order_by()`                   | сортировка                         |
| `.limit()`                      | ограничение количества результатов |
| `.options(selectinload(...))`   | подгрузка связанных объектов       |
